# 🔊 Best Methods for Phonetic Pronunciation (TTS)

## Problem

Current phonetic guides like "keels", "ah-vahyant" are being **read as words** instead of **pronounced phonetically** by TTS engines.

Example:
- TTS says: "keels" (like the word "keels")
- Should say: "k-ee-l-s" (phonetic breakdown)

---

## ✅ Solution 1: Use IPA (International Phonetic Alphabet) + SSML (Recommended)

This is the **most accurate** method. Convert your phonetics to IPA, then use SSML with Web Speech API.

### Step 1: Convert to IPA

```typescript
// server/lib/phonetic-converter.ts

const PHONETIC_TO_IPA_MAP: Record<string, string> = {
  // Vowels
  'ah': 'ɑ',
  'ay': 'eɪ',
  'ee': 'i',
  'eh': 'ɛ',
  'ih': 'ɪ',
  'oh': 'oʊ',
  'oo': 'u',
  'uh': 'ʌ',
  'aw': 'ɔ',
  
  // Consonants
  'ch': 'tʃ',
  'sh': 'ʃ',
  'th': 'θ',
  'dh': 'ð',
  'ng': 'ŋ',
  'zh': 'ʒ',
  
  // R-colored vowels
  'er': 'ɜr',
  'ar': 'ɑr',
  'or': 'ɔr',
  'air': 'ɛr',
  
  // Diphthongs
  'oy': 'ɔɪ',
  'ow': 'aʊ',
  
  // Common syllables
  'vahyant': 'væjənt',
  'proh': 'proʊ',
  'fohnd': 'fɔnd',
  'uhr': 'ər',
  'nohs': 'noʊz',
  'seeuhx': 'sjø'
};

export function convertToIPA(phonetic: string): string {
  let ipa = phonetic.toLowerCase();
  
  // Split by hyphens and spaces
  const parts = ipa.split(/[-\s]+/);
  
  // Convert each part
  const converted = parts.map(part => {
    return PHONETIC_TO_IPA_MAP[part] || part;
  });
  
  return converted.join('');
}

// Example usage:
convertToIPA('keels');          // 'kilz'
convertToIPA('ah-vahyant');     // 'ɑvæjənt'
convertToIPA('proh-fohnd-uhr'); // 'proʊfɔndər'
```

### Step 2: Use Web Speech API with IPA

```typescript
// client/src/lib/phonetic-tts.ts

export class PhoneticTTS {
  private synth = window.speechSynthesis;
  private voice: SpeechSynthesisVoice | null = null;

  constructor() {
    this.loadVoice();
  }

  private loadVoice() {
    const voices = this.synth.getVoices();
    
    // Prefer high-quality voices that support IPA
    const preferredVoices = [
      'Google US English',
      'Microsoft Zira',
      'Apple Samantha',
      'Google français'
    ];

    for (const voiceName of preferredVoices) {
      const voice = voices.find(v => v.name.includes(voiceName));
      if (voice) {
        this.voice = voice;
        break;
      }
    }

    // Fallback to first available voice
    if (!this.voice && voices.length > 0) {
      this.voice = voices[0];
    }
  }

  async speakPhonetic(phoneticText: string, language: string = 'en-US') {
    return new Promise<void>((resolve, reject) => {
      // Convert to IPA
      const ipaText = this.convertToIPA(phoneticText);
      
      console.log('[TTS] Original:', phoneticText);
      console.log('[TTS] IPA:', ipaText);

      const utterance = new SpeechSynthesisUtterance(ipaText);
      
      // Configure for phonetic pronunciation
      utterance.voice = this.voice;
      utterance.lang = language;
      utterance.rate = 0.75;  // Slower for clarity
      utterance.pitch = 1.0;
      utterance.volume = 1.0;

      utterance.onend = () => resolve();
      utterance.onerror = (error) => reject(error);

      this.synth.speak(utterance);
    });
  }

  private convertToIPA(phonetic: string): string {
    // Simple conversion - expand this based on your needs
    return phonetic
      .replace(/ah/g, 'ɑ')
      .replace(/ay/g, 'eɪ')
      .replace(/ee/g, 'i')
      .replace(/oh/g, 'oʊ')
      .replace(/oo/g, 'u')
      .replace(/uh/g, 'ʌ')
      .replace(/-/g, ''); // Remove hyphens
  }
}
```

## ✅ Solution 2: Use Google Cloud Text-to-Speech with SSML (Best Quality)

Google's TTS has the best IPA support and pronunciation control.

### Setup

```bash
npm install @google-cloud/text-to-speech
```

```typescript
// server/lib/google-tts.ts

import textToSpeech from '@google-cloud/text-to-speech';

const client = new textToSpeech.TextToSpeechClient({
  keyFilename: process.env.GOOGLE_APPLICATION_CREDENTIALS
});

export async function generatePhoneticAudio(
  phoneticText: string,
  language: string = 'en-US'
): Promise<Buffer> {
  // Convert to IPA
  const ipaText = convertToIPA(phoneticText);

  // Create SSML with phoneme tags
  const ssml = `
    <speak>
      <phoneme alphabet="ipa" ph="${ipaText}">
        ${phoneticText}
      </phoneme>
    </speak>
  `;

  console.log('[Google TTS] SSML:', ssml);

  const [response] = await client.synthesizeSpeech({
    input: { ssml },
    voice: {
      languageCode: language,
      name: `${language}-Wavenet-D` // High quality voice
    },
    audioConfig: {
      audioEncoding: 'MP3',
      speakingRate: 0.75, // Slower for learning
      pitch: 0,
      volumeGainDb: 0
    }
  });

  return Buffer.from(response.audioContent as Uint8Array);
}
```

### Backend Route

```typescript
// server/routes.ts

app.post('/api/tts/phonetic', async (req, res) => {
  try {
    const { phoneticText, language } = req.body;

    const audioBuffer = await generatePhoneticAudio(phoneticText, language);

    res.set({
      'Content-Type': 'audio/mpeg',
      'Content-Length': audioBuffer.length
    });

    res.send(audioBuffer);

  } catch (error) {
    console.error('[TTS] Error:', error);
    res.status(500).json({ error: 'Failed to generate audio' });
  }
});
```

### Frontend Usage

```typescript
// client/src/components/word-practice.tsx

async function playPhonetic(phoneticText: string) {
  try {
    const response = await fetch('/api/tts/phonetic', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        phoneticText,
        language: 'fr-FR' // or 'en-US', etc.
      })
    });

    const audioBlob = await response.blob();
    const audioUrl = URL.createObjectURL(audioBlob);
    
    const audio = new Audio(audioUrl);
    await audio.play();

  } catch (error) {
    console.error('[TTS] Failed to play:', error);
  }
}
```

## ✅ Solution 3: Use ElevenLabs API (Best Natural Sound)

ElevenLabs has the most natural-sounding voices and good phonetic control.

```bash
npm install elevenlabs
```

```typescript
// server/lib/elevenlabs-tts.ts

import { ElevenLabsClient } from 'elevenlabs';

const elevenlabs = new ElevenLabsClient({
  apiKey: process.env.ELEVENLABS_API_KEY
});

export async function generatePhoneticAudio(
  phoneticText: string,
  language: string = 'en'
): Promise<Buffer> {
  const ipaText = convertToIPA(phoneticText);

  const audio = await elevenlabs.generate({
    voice: 'Rachel', // Or other voice
    text: ipaText,
    model_id: 'eleven_multilingual_v2'
  });

  const chunks: Buffer[] = [];
  for await (const chunk of audio) {
    chunks.push(chunk);
  }

  return Buffer.concat(chunks);
}
```

For common phonetic patterns, pre-record audio files:

```typescript
// server/lib/phonetic-audio-cache.ts

const PHONETIC_AUDIO_MAP: Record<string, string> = {
```

## ✅ Solution 5: Syllable-by-Syllable Playback

Break down phonetics and play each syllable:

```typescript
// client/src/lib/syllable-tts.ts

export async function speakPhoneticBySyllable(phoneticText: string) {
  const syllables = phoneticText.split('-');
  
  for (const syllable of syllables) {
    // Play each syllable with a pause
    await speakSingleSyllable(syllable);
    await sleep(300); // 300ms pause between syllables
  }
}

async function speakSingleSyllable(syllable: string): Promise<void> {
  return new Promise((resolve) => {
    const utterance = new SpeechSynthesisUtterance(syllable);
    utterance.rate = 0.6; // Very slow
    utterance.pitch = 1.2; // Slightly higher pitch
    
    utterance.onend = () => resolve();
    
    window.speechSynthesis.speak(utterance);
  });
}
```

## 🎯 Recommended Approach

**Use a hybrid approach:**

1. **Pre-record common phonetic patterns** (saves API costs)
2. **Google Cloud TTS with SSML** for dynamic content
3. **Web Speech API** as fallback

```typescript
// client/src/lib/phonetic-player.ts

export class PhoneticPlayer {
  async play(phoneticText: string, language: string) {
    // 1. Try pre-recorded audio
    const preRecorded = await this.tryPreRecorded(phoneticText);
    if (preRecorded) return;

    // 2. Try Google Cloud TTS
    try {
      await this.playGoogleTTS(phoneticText, language);
      return;
    } catch (error) {
      console.error('[TTS] Google TTS failed:', error);
    }

    // 3. Fallback to Web Speech API
    await this.playWebSpeech(phoneticText, language);
  }

  private async tryPreRecorded(phoneticText: string): Promise<boolean> {
    const url = PHONETIC_AUDIO_MAP[phoneticText.toLowerCase()];
    if (!url) return false;

    try {
      const audio = new Audio(url);
      await audio.play();
      return true;
    } catch {
      return false;
    }
  }

  private async playGoogleTTS(phoneticText: string, language: string) {
    const response = await fetch('/api/tts/phonetic', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ phoneticText, language })
    });

    const blob = await response.blob();
    const audio = new Audio(URL.createObjectURL(blob));
    await audio.play();
  }

  private async playWebSpeech(phoneticText: string, language: string) {
    const ipaText = convertToIPA(phoneticText);
    
    return new Promise<void>((resolve) => {
      const utterance = new SpeechSynthesisUtterance(ipaText);
      utterance.lang = language;
      utterance.rate = 0.75;
      utterance.onend = () => resolve();
      
      window.speechSynthesis.speak(utterance);
    });
  }
}
```
